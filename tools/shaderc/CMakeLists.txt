add_executable(truffle-shaderc main.cpp)
target_link_libraries(truffle-shaderc PRIVATE truffle_shader)
target_compile_features(truffle-shaderc PRIVATE cxx_std_20)
truffle_target_warnings(truffle-shaderc)
if(UNIX AND (TRUFFLE_SHADER_ENABLE_SLANG OR TRUFFLE_SHADER_ENABLE_DXC))
    file(RELATIVE_PATH _truffle_shaderc_runtime_path
        "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
    if(IS_ABSOLUTE "${CMAKE_INSTALL_LIBDIR}")
        set(_truffle_shaderc_install_rpath "${CMAKE_INSTALL_LIBDIR}")
    elseif(APPLE)
        set(_truffle_shaderc_install_rpath "@loader_path/${_truffle_shaderc_runtime_path}")
    else()
        set(_truffle_shaderc_install_rpath "$ORIGIN/${_truffle_shaderc_runtime_path}")
    endif()
    set_property(TARGET truffle-shaderc APPEND PROPERTY INSTALL_RPATH
        "${_truffle_shaderc_install_rpath}")
endif()

if(TRUFFLE_INSTALL)
    install(TARGETS truffle-shaderc RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
